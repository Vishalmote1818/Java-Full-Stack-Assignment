-- Display all order of customer from punne ..

SELECT * FROM orders WHERE city='Pune';


-- Show all orders where price greater than 5000

SELECT * FROM orders WHERE price> 5000;

-- Sellect all orders where payment mode is UPI

SELECT * FROM orders WHERE payment_mode='upi';

-- Display all orders where order_status is delivered

SELECT * FROM orders WHERE order_status='Delivered';

-- show orders where quantity is between 2 and 5

SELECT * FROM orders WHERE quantity BETWEEN 2 AND 5;

-- Display orders where category is electronics or clothing

SELECT * FROM orders 
WHERE category = 'Electronics' OR category = 'Clothing';


-- Find orders where customer name starts with A 

SELECT * FROM orders WHERE customer_name LIKE 'A%';

-- Show orders where product name contains phone

SELECT * FROM orders WHERE product_name LIKE '%Phone%';


-- Display all orders sorted by price in descending order

SELECT * FROM orders ORDER BY Price DESC;

-- Show orders sorted by order date(latest first)alter

SELECT * FROM orders ORDER BY order_date DESC;

-- Find total revenue (Price * quantity)

SELECT order_id, (Price*quantity) AS Total_Price FROM orders ;

-- Average price of product

SELECT AVG(Price) AS Average_Price FROM orders;

-- Count how many oders are placed;

SELECT COUNT(order_id) AS Total_Order FROM orders;
